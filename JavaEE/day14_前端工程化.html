import {defineStore} from 'pinia'
/* 
    id: 当前对象的唯一标识
    state: 设置共享数据
    getters: 取值(支持运算)
    actions: 操作
*/
//第一种写法
/* let defineUser666=defineStore({
    id:'user101',
    state:()=>{
        return {

            age:'20',
            hobby:['java','唱','跳']
        }
    },
    getters:{
        getHobbyCount(){
            return this.hobby.length
        },
        getAge(){
            return this.age
        }
    },
    actions:{
        doubleAge(){
            this.age*=2
        },
    }
}) */
//第二种写法
let defineUser666=defineStore('user102',{
    state:()=>{
        return {
            id:1,
            username:'admin',
            password:'root',
            age:'20',
            hobby:['java','唱','跳']
        }
    },
    getters:{
        getHobbyCount(){
            return this.hobby.length
        },
        getAge(){
            return this.age
        },
        getCount:(state)=>{
            return state.hobby.length
        }
    },
    actions:{
        doubleAge(){
            this.age*=2
        },
    }
})

export {defineUser666}

    # 4.2 展示我的日程信息

    • 在showSchedule页面挂载完成时，发送异步请求到服务器

      • 携带当前登录人的uid

    • 服务器就接收请求，获取uid,调用service,service调dao,查到当前用户的日程信息(List<SysSchedule>)

    • servlet将list集合放到Result的data属性上，将Result对象写回前端

    • 前段拿到数据，赋值给数组或pinia，数组或pinia和页面上的表格进行的列表渲染
